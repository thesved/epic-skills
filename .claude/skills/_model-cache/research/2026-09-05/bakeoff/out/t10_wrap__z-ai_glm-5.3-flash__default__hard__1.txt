def fill(text: str, width: int) -> str:
    if width <= 0:
        raise ValueError("invalid width %r (must be > 0)" % (width,))

    # --- 1. Munge whitespace ---------------------------------------------
    # Expand tabs, then replace the whitespace characters recognized by
    # string.whitespace with plain spaces.
    text = text.expandtabs()
    text = text.translate({ord(c): ' ' for c in '\t\n\x0b\x0c\r '})

    # --- 2. Split into indivisible chunks (wordsep_re behaviour) ---------
    # Chunks are: runs of whitespace, runs of word characters (\w), and
    # maximal runs of other non-whitespace characters.  Hyphenated words
    # therefore get split around each hyphen ('foo-bar' -> 'foo', '-', 'bar')
    # and every boundary is a potential break point.
    chunks = []
    i, n = 0, len(text)
    while i < n:
        if text[i].isspace():
            j = i + 1
            while j < n and text[j].isspace():
                j += 1
            chunks.append(text[i:j])
            i = j
            continue
        while i < n and not text[i].isspace():
            j = i
            if text[i].isalnum() or text[i] == '_':
                # run of word characters
                while j < n and (text[j].isalnum() or text[j] == '_'):
                    j += 1
            else:
                # run of non-word, non-whitespace characters
                # (punctuation, hyphens, quotes, ...)
                while j < n and not text[j].isspace() and \
                        not (text[j].isalnum() or text[j] == '_'):
                    j += 1
            chunks.append(text[i:j])
            i = j

    # --- 3. Greedily assemble lines from the chunks -----------------------
    lines = []
    chunks.reverse()  # use list end as the "next" chunk
    while chunks:
        cur_line = []
        cur_len = 0

        # Drop leading whitespace on continuation lines (not at the very
        # beginning of the text).
        if chunks[-1].strip() == '' and lines:
            del chunks[-1]

        while chunks:
            l = len(chunks[-1])
            if cur_len + l <= width:
                cur_line.append(chunks.pop())
                cur_len += l
            else:
                break

        # The line is full and the next chunk is too big to fit on any
        # line: slice off as much as fits (break_long_words=True).
        if chunks and len(chunks[-1]) > width:
            space_left = 1 if width < 1 else width - cur_len
            cur_line.append(chunks[-1][:space_left])
            chunks[-1] = chunks[-1][space_left:]

        # Drop trailing whitespace from the finished line.
        if cur_line and cur_line[-1].strip() == '':
            del cur_line[-1]

        if cur_line:
            lines.append(''.join(cur_line))

    return '\n'.join(lines)
