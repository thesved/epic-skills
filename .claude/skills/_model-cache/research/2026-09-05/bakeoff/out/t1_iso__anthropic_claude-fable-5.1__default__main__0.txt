def parse_duration(s: str) -> float:
    if not isinstance(s, str):
        raise ValueError("duration must be a string")
    if len(s) < 2 or s[0] != 'P':
        raise ValueError("invalid ISO 8601 duration: %r" % (s,))

    body = s[1:]
    n = len(body)
    i = 0

    date_units = {'Y': 365 * 86400.0, 'M': 30 * 86400.0, 'W': 7 * 86400.0, 'D': 86400.0}
    time_units = {'H': 3600.0, 'M': 60.0, 'S': 1.0}
    date_order = ['Y', 'M', 'W', 'D']
    time_order = ['H', 'M', 'S']
    digits = '0123456789'

    in_time = False
    last_idx = -1
    seen_fraction = False
    total = 0.0
    count = 0
    time_count = 0
    has_w = False

    while i < n:
        c = body[i]
        if c == 'T':
            if in_time:
                raise ValueError("invalid ISO 8601 duration: %r" % (s,))
            in_time = True
            last_idx = -1
            i += 1
            continue
        if c not in digits:
            raise ValueError("invalid ISO 8601 duration: %r" % (s,))
        start = i
        while i < n and body[i] in digits:
            i += 1
        int_part = body[start:i]
        frac_part = None
        if i < n and body[i] in '.,':
            i += 1
            fs = i
            while i < n and body[i] in digits:
                i += 1
            frac_part = body[fs:i]
            if not frac_part:
                raise ValueError("invalid ISO 8601 duration: %r" % (s,))
        if i >= n:
            raise ValueError("invalid ISO 8601 duration: %r" % (s,))
        letter = body[i]
        i += 1

        if seen_fraction:
            raise ValueError("invalid ISO 8601 duration: %r" % (s,))
        if frac_part is not None:
            seen_fraction = True
            value = float(int_part + '.' + frac_part)
        else:
            value = float(int_part)

        if in_time:
            if letter not in time_units:
                raise ValueError("invalid ISO 8601 duration: %r" % (s,))
            idx = time_order.index(letter)
            if idx <= last_idx:
                raise ValueError("invalid ISO 8601 duration: %r" % (s,))
            last_idx = idx
            time_count += 1
            total += value * time_units[letter]
        else:
            if letter not in date_units:
                raise ValueError("invalid ISO 8601 duration: %r" % (s,))
            idx = date_order.index(letter)
            if idx <= last_idx:
                raise ValueError("invalid ISO 8601 duration: %r" % (s,))
            last_idx = idx
            if letter == 'W':
                has_w = True
            total += value * date_units[letter]
        count += 1

    if count == 0:
        raise ValueError("invalid ISO 8601 duration: %r" % (s,))
    if in_time and time_count == 0:
        raise ValueError("invalid ISO 8601 duration: %r" % (s,))
    if has_w and (count > 1 or in_time):
        raise ValueError("invalid ISO 8601 duration: %r" % (s,))

    return float(total)
