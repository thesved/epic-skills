```python
import re

_wordsep_re = re.compile(
    r'(\s+|'
    r'[^\s\w]*\w+[^0-9\W]-(?=\w+[^0-9\W])|'
    r'(?<=[\w!"\'&.,?])-{2,}(?=\w))'
)

_ws_map = {ord(c): ord(' ') for c in '\t\n\x0b\x0c\r'}

def _munge(text):
    text = text.expandtabs(8)
    text = text.translate(_ws_map)
    return text

def _split(text):
    chunks = _wordsep_re.split(text)
    return [c for c in chunks if c]

def fill(text: str, width: int) -> str:
    if width <= 0:
        raise ValueError("invalid width %r (must be > 0)" % (width,))
    text = _munge(text)
    chunks = _split(text)
    chunks.reverse()
    lines = []
    while chunks:
        cur_line = []
        cur_len = 0
        if chunks[-1].strip() == '' and lines:
            del chunks[-1]
        while chunks:
            l = len(chunks[-1])
            if cur_len + l <= width:
                cur_line.append(chunks.pop())
                cur_len += l
            else:
                break
        if chunks and len(chunks[-1]) > width:
            if width < 1:
                space_left = 1
            else:
                space_left = width - cur_len
            lchunk = chunks[-1]
            cur_line.append(lchunk[:space_left])
            chunks[-1] = lchunk[space_left:]
        if cur_line and cur_line[-1].strip() == '':
            del cur_line[-1]
        if cur_line:
            lines.append(''.join(cur_line))
    return '\n'.join(lines)
```
