import re

def parse_duration(s: str) -> float:
    if not isinstance(s, str) or not s.startswith('P') or len(s) <= 1:
        raise ValueError("Invalid ISO 8601 duration format")

    body = s[1:]

    if 'W' in body:
        m = re.fullmatch(r'(\d+(?:[.,]\d+)?)W', body)
        if not m:
            raise ValueError("Invalid week duration")
        val = float(m.group(1).replace(',', '.'))
        return float(val * 7 * 86400)

    date_part, sep, time_part = body.partition('T')

    if sep and not time_part:
        raise ValueError("T must be followed by at least one time component")

    if not date_part and not time_part:
        raise ValueError("Empty duration")

    def parse_section(part: str, allowed_units: list[tuple[str, float]]) -> list[tuple[float, bool, float]]:
        pos = 0
        unit_idx = 0
        results = []
        pattern = re.compile(r'(\d+(?:[.,]\d+)?)([A-Za-z])')
        for m in pattern.finditer(part):
            if m.start() != pos:
                raise ValueError(f"Invalid characters in duration: {part}")
            pos = m.end()
            val_str, unit = m.groups()
            found = False
            for i in range(unit_idx, len(allowed_units)):
                if allowed_units[i][0] == unit:
                    unit_idx = i + 1
                    found = True
                    has_fraction = '.' in val_str or ',' in val_str
                    val = float(val_str.replace(',', '.'))
                    results.append((val, has_fraction, allowed_units[i][1]))
                    break
            if not found:
                raise ValueError(f"Unit {unit} is invalid, duplicated, or out of order")
        if pos != len(part) or not results:
            raise ValueError(f"Invalid duration section: {part}")
        return results

    all_components = []

    if date_part:
        date_units = [('Y', 365 * 86400), ('M', 30 * 86400), ('D', 86400)]
        all_components.extend(parse_section(date_part, date_units))

    if time_part:
        time_units = [('H', 3600), ('M', 60), ('S', 1)]
        all_components.extend(parse_section(time_part, time_units))

    for i, (_, has_fraction, _) in enumerate(all_components):
        if has_fraction and i != len(all_components) - 1:
            raise ValueError("Only the last component may contain a decimal fraction")

    total_seconds = sum(val * mult for val, _, mult in all_components)
    return float(total_seconds)
